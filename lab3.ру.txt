# Лабораторная работа №3. Вариант 1: Таблица умножения + сумма цифр числа
print("--- Задание 1: Таблица умножения ---")

# Цикл for для вывода таблицы умножения
for i in range(1, 11):
    print(f"5 x {i} = {5 * i}")

print("\n--- Задание 2: Сумма цифр числа ---")

# Ввод числа (проверяем на ошибки)
try:
    number = int(input("Введите целое число: "))
    
    # Берем модуль числа, если ввели отрицательное
    number = abs(number) 
    
    sum_digits = 0
    
    # Цикл while для подсчета суммы цифр
    # Мы берем остаток от деления на 10 (последнюю цифру) и прибавляем к сумме,
    # а затем делим число на 10 нацело, отбрасывая последнюю цифру.
    while number > 0:
        digit = number % 10      # Получаем последнюю цифру
        sum_digits += digit      # Добавляем к сумме
        number = number // 10    # Убираем последнюю цифру
        
    print(f"Сумма цифр введенного числа: {sum_digits}")

except ValueError:
    print("Ошибка: вы ввели не целое число!")
