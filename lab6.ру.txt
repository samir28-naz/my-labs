import math

def area_circle(radius):
    return math.pi * (radius ** 2)

def area_rectangle(a, b):
    return a * b

def area_triangle(a, b, c):
    p = (a + b + c) / 2
    return math.sqrt(p * (p - a) * (p - b) * (p - c))

def main():
    while True:
        print("\n--- МЕНЮ ---")
        print("1. Площадь круга")
        print("2. Площадь прямоугольника")
        print("3. Площадь треугольника")
        print("0. Выход")
        
        choice = input("Выберите пункт: ")
        
        try:
            if choice == '1':
                r = float(input("Введите радиус: "))
                print("Площадь круга:", round(area_circle(r), 2))
            elif choice == '2':
                a = float(input("Введите сторону a: "))
                b = float(input("Введите сторону b: "))
                print("Площадь прямоугольника:", round(area_rectangle(a, b), 2))
            elif choice == '3':
                a = float(input("Введите сторону a: "))
                b = float(input("Введите сторону b: "))
                c = float(input("Введите сторону c: "))
                print("Площадь треугольника:", round(area_triangle(a, b, c), 2))
            elif choice == '0':
                print("Выход из программы.")
                break
            else:
                print("Неверный пункт меню!")
        except ValueError:
            print("Ошибка: вводите только числа!")
        except Exception as e:
            print("Ошибка вычислений:", e)

if __name__ == "__main__":
    main()
