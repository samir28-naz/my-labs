print("--- Определение високосности года и количества дней ---")


try:
    year = int(input("Введите год: "))
    
    # Проверка на корректность ввода (год не может быть отрицательным или нулевым)
    if year <= 0:
        print("Ошибка: год должен быть положительным числом!")
    else:
       
        if (year % 4 == 0 and year % 100 != 0) or (year % 400 == 0):
            print(f"Год {year} — ВИСОКОСНЫЙ.")
            print("Количество дней в году: 366")
        else:
            print(f"Год {year} — НЕ ВИСОКОСНЫЙ.")
            print("Количество дней в году: 365")
            
except ValueError:
    print("Ошибка: вы ввели не число! Пожалуйста, запустите программу заново.")
